% Symbols and notation used in this dissertation
\begin{dictionary}{Glossary}
\item[$N$] Number of residual blocks in a network
\item[$F$] Number of convolutional filters per layer
\item[$d$] Embedding dimension (Transformer models)
\item[$h$] Number of attention heads
\item[$L$] Number of Transformer encoder layers
\item[$\pi(a \mid s)$] Policy: probability of selecting move $a$ in board state $s$
\item[$v(s)$] Value: predicted game outcome from board state $s$
\item[$\mathbf{X}$] Node feature matrix in GNN input
\item[$\mathbf{A}$] Adjacency matrix representing piece connectivity or legal moves
\item[cp] Centipawn; one hundredth of a pawn's value, used by engines to score positions
\item[ELO] Rating system for measuring relative playing strength
\item[ply] A half-move; one move by either White or Black
\end{dictionary}
